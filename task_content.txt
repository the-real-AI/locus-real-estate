Шаг 1. Модель и DTO (0:00–0:30)
Сущность — то, что сервер хранит. DTO (Data Transfer Object) — то, что ходит по сети. Разделяем сразу, чтобы клиент не мог задать Id и дату создания, а лишние поля не утекали наружу.
Models/Listing.cs
namespace ListingsApi.Models;
 
public class Listing
{
    public int Id { get; set; }
    public string Title { get; set; } = "";
    public decimal Price { get; set; }        // в условных единицах
    public string District { get; set; } = "";
    public int Rooms { get; set; }
    public DateTime CreatedAt { get; set; }
}
Contracts/ListingContracts.cs
namespace ListingsApi.Contracts;
 
// то, что клиент присылает при создании
public record CreateListingRequest(string Title, decimal Price, string District, int Rooms);
 
// то, что клиент присылает при полной замене (PUT)
public record UpdateListingRequest(string Title, decimal Price, string District, int Rooms);
 
// то, что клиент получает
public record ListingResponse(int Id, string Title, decimal Price, string District, int Rooms, DateTime CreatedAt);
record — тип «только данные»: конструктор, свойства, сравнение и ToString C# генерирует сам. Для DTO это идеальный выбор. Create и Update сейчас одинаковые — оставляем два record'а намеренно: на уроке 6 у них появятся разные правила валидации.
Шаг 2. Хранилище в памяти и DI (0:30–0:50)
Хранилище — обычный класс со списком. Регистрируем его в контейнере DI как Singleton (один экземпляр на всё приложение), и любой endpoint получит его, просто указав в параметрах.
Services/ListingStore.cs
using ListingsApi.Contracts;
using ListingsApi.Models;
 
namespace ListingsApi.Services;
 
public class ListingStore
{
    private readonly List<Listing> _items = new();
    private readonly object _lock = new();
    private int _nextId = 1;
 
    public ListingStore()
    {
        // стартовые данные, чтобы список не был пустым
        Add(new CreateListingRequest("2-комн. рядом с метро Чиланзар", 48000, "Чиланзар", 2));
        Add(new CreateListingRequest("3-комн. с ремонтом", 71000, "Юнусабад", 3));
        Add(new CreateListingRequest("1-комн. студия", 32000, "Мирзо-Улугбек", 1));
    }
 
    public IReadOnlyList<Listing> GetAll()
    {
        lock (_lock) return _items.ToList();
    }
 
    public Listing? Get(int id)
    {
        lock (_lock) return _items.Find(l => l.Id == id);
    }
 
    public Listing Add(CreateListingRequest r)
    {
        lock (_lock)
        {
            var listing = new Listing
            {
                Id = _nextId++, Title = r.Title, Price = r.Price,
                District = r.District, Rooms = r.Rooms, CreatedAt = DateTime.UtcNow
            };
            _items.Add(listing);
            return listing;
        }
    }
 
    public bool Update(int id, UpdateListingRequest r)
    {
        lock (_lock)
        {
            var listing = _items.Find(l => l.Id == id);
            if (listing is null) return false;
            listing.Title = r.Title; listing.Price = r.Price;
            listing.District = r.District; listing.Rooms = r.Rooms;
            return true;
        }
    }
 
    public bool Delete(int id)
    {
        lock (_lock) return _items.RemoveAll(l => l.Id == id) > 0;
    }
}
Регистрация в 
Program.cs
 — одна строка до 
builder.Build()
:
builder.Services.AddSingleton<ListingStore>();
Зачем lock: сервер обрабатывает запросы параллельно. Два одновременных POST без блокировки могут получить один и тот же Id. Глубже в многопоточность сегодня не идём.
Шаг 3. Чтение: GET списка и GET по id (0:50–1:15)
Все endpoint'ы кладём в отдельный файл как метод-расширение: Program.cs останется коротким. Начинаем с двух GET.
Endpoints/ListingEndpoints.cs
using ListingsApi.Contracts;
using ListingsApi.Models;
using ListingsApi.Services;
using Microsoft.AspNetCore.Http.HttpResults;
 
namespace ListingsApi.Endpoints;
 
public static class ListingEndpoints
{
    public static void MapListingEndpoints(this WebApplication app)
    {
        var group = app.MapGroup("/api/listings").WithTags("Listings");
 
        // GET /api/listings?district=Чиланзар&maxPrice=50000
        group.MapGet("/", (ListingStore store, string? district, decimal? maxPrice) =>
        {
            var items = store.GetAll().AsEnumerable();
            if (district is not null) items = items.Where(l => l.District == district);
            if (maxPrice is not null) items = items.Where(l => l.Price <= maxPrice);
            return TypedResults.Ok(items.Select(ToResponse));
        })
        .WithName("GetListings")
        .WithSummary("Список объявлений с фильтром по району и максимальной цене");
 
        // GET /api/listings/7
        group.MapGet("/{id:int}", Results<Ok<ListingResponse>, NotFound> (int id, ListingStore store) =>
            store.Get(id) is { } listing
                ? TypedResults.Ok(ToResponse(listing))
                : TypedResults.NotFound())
        .WithName("GetListing")
        .WithSummary("Одно объявление по id");
    }
 
    private static ListingResponse ToResponse(Listing l) =>
        new(l.Id, l.Title, l.Price, l.District, l.Rooms, l.CreatedAt);
}

Шаг 4. Создание: POST → 201 Created (1:15–1:45)
Добавьте в MapListingEndpoints после второго GET:
        // POST /api/listings
        group.MapPost("/", Results<Created<ListingResponse>, BadRequest<string>>
            (CreateListingRequest request, ListingStore store) =>
        {
            if (string.IsNullOrWhiteSpace(request.Title))
                return TypedResults.BadRequest("Поле title обязательно");
            if (request.Price <= 0)
                return TypedResults.BadRequest("Поле price должно быть больше 0");
            if (request.Rooms is < 1 or > 10)
                return TypedResults.BadRequest("Поле rooms — от 1 до 10");
 
            var created = store.Add(request);
            return TypedResults.Created($"/api/listings/{created.Id}", ToResponse(created));
        })
        .WithName("CreateListing")
        .WithSummary("Создать объявление");
Что здесь важно: 
CreateListingRequest request
 — сложный тип, значит ASP.NET возьмёт его из JSON-тела запроса (привязка модели). 
TypedResults.Created(url, dto)
 вернёт код 201, тело с созданным объектом и заголовок 
Location
 с адресом нового ресурса — клиент знает, куда идти за ним.
Проверки «руками» — временные. На уроке 6 их заменит настоящая валидация с ошибками по каждому полю в формате ProblemDetails. Сейчас важно понять сам принцип: неверный ввод → 400, а не 500.
Проверка через браузер невозможна
 — адресная строка умеет только GET. Переходите к шагу 6 (Swagger) или сразу создайте файл из шага 7 и отправьте POST оттуда.
Шаг 5. Изменение и удаление: PUT и DELETE → 204 (1:45–2:15)
        // PUT /api/listings/7 — полная замена
        group.MapPut("/{id:int}", Results<NoContent, NotFound, BadRequest<string>>
            (int id, UpdateListingRequest request, ListingStore store) =>
        {
            if (string.IsNullOrWhiteSpace(request.Title) || request.Price <= 0)
                return TypedResults.BadRequest("title обязателен, price > 0");
            return store.Update(id, request)
                ? TypedResults.NoContent()
                : TypedResults.NotFound();
        })
        .WithName("UpdateListing")
        .WithSummary("Заменить объявление целиком");
 
        // DELETE /api/listings/7
        group.MapDelete("/{id:int}", Results<NoContent, NotFound> (int id, ListingStore store) =>
            store.Delete(id) ? TypedResults.NoContent() : TypedResults.NotFound())
        .WithName("DeleteListing")
        .WithSummary("Удалить объявление");
Здесь два параметра из разных мест: 
int id
 — из пути, 
UpdateListingRequest request
 — из тела. Ответ 204 No Content означает «получилось, возвращать нечего». Обратите внимание, что 
Results<…>
 перечисляет все три возможных исхода — Swagger покажет их автоматически.

в Program.cs сейчас должно быть примерно так:

using ListingsApi.Endpoints;

var builder = WebApplication.CreateBuilder(args);
 
builder.Services.AddOpenApi();

Шаг 6. OpenAPI и Swagger UI (2:15–2:45)
Документ OpenAPI ASP.NET Core уже генерирует: откройте 
http://localhost:5000/openapi/v1.json
 и найдите там

Остановите сервер (Ctrl+C) и в папке API выполните:

В

 сразу после

{
    options.SwaggerEndpoint("/openapi/v1.json", "Listings API v1");
    options.RoutePrefix = "swagger";      // страница будет по адресу /swagger
});
Запустите 
dotnet run
 и откройте 
http://localhost:5000/swagger
. Должны быть видны пять методов в группе Listings с вашими описаниями из 
WithSummary
.
Проверьте все пять: раскройте метод → 
Try it out
 → заполните → 
Execute
. Для POST используйте тело:
{
  "title": "4-комн. с видом на парк",
  "price": 95000,
  "district": "Яккасарай",
  "rooms": 4
}
В ответе на POST найдите код 201 и в блоке Response headers — строку

. Затем сделайте GET по этому id, PUT с изменённой ценой, DELETE и снова GET — должен быть 404.
Альтернатива Swagger UI — Scalar: dotnet add package Scalar.AspNetCore, в Program.cs app.MapScalarApiReference(), адрес /scalar. Читает тот же документ, выглядит современнее. Выбор на ваш вкус, в отчёте укажите, что использовали.
В старых уроках и на YouTube встречается AddSwaggerGen() / UseSwagger(). В .NET 10 они не нужны: документ делает встроенный AddOpenApi, а Swashbuckle используется только для страницы.

Шаг 8. Angular: список из настоящего API (3:10–3:35)
Небольшой шаг — заменяем строку из урока 1 на список объявлений. Полноценная работа с сервисами, маршрутизацией и состояниями загрузки — урок 5.
src/app/listing.ts
// поля в camelCase — ровно так, как сервер отдаёт JSON
export interface Listing {
  id: number;
  title: string;
  price: number;
  district: string;
  rooms: number;
  createdAt: string;
}
src/app/app.ts
import { Component, inject, signal } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Listing } from './listing';
 
@Component({
  selector: 'app-root',
  template: `
    <h1>Объявления</h1>
    @if (error()) {
      <p class="error">{{ error() }}</p>
    }
    <ul>
      @for (item of listings(); track item.id) {
        <li>{{ item.title }} — {{ item.district }}, {{ item.rooms }} комн., {{ item.price }} у.е.</li>
      } @empty {
        <li>Пока пусто</li>
      }
    </ul>
  `,
})
export class App {
  private http = inject(HttpClient);
  listings = signal<Listing[]>([]);
  error = signal<string | null>(null);
 
  constructor() {
    this.http.get<Listing[]>('http://localhost:5000/api/listings').subscribe({
      next: (data) => this.listings.set(data),
      error: (err) => this.error.set('Сервер недоступен: ' + err.message),
    });
  }
}
